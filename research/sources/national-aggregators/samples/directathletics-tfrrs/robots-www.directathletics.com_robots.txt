    var rr = 0;
    function next_info() {
        fade('faded', 1, 0.0, 800, new Date().getTime());
        setTimeout(next_info_helper, 800);

    }
    function next_info_helper () {
        var target = document.getElementById('faded');
        var arrow;
        if(rr == 0) {
            rr = 1;
            arrow = document.getElementById('tfrrs_info');
        } else if (rr == 1) {
            rr = 2;
            arrow = document.getElementById('mp_info');
        } else {
            rr = 0;
            arrow = document.getElementById('da_info');
        }
        target.innerHTML = arrow.innerHTML;
        fade('faded', 0.0, 1, 800, new Date().getTime());
        setTimeout("fade( '" + 'faded' + "'," + '0.0' + "," + '1' + "," + '800' + "," + (new Date().getTime()) + ")", 800);
        setTimeout(next_info, 7000);
    }

    function fade(eid, initOp, finalOp, TimeToFade, time) {
    if (initOp==0)
    {
        document.getElementById(eid).style.visibility="visible";
    }
    var curTick = new Date().getTime();
    var elapsedTicks = curTick - time;
    var newOp = initOp+(finalOp-initOp)*elapsedTicks/TimeToFade;
    if (Math.abs(newOp-initOp)>Math.abs(finalOp-initOp))
    {
        document.getElementById(eid).style.opacity=finalOp;
        if (finalOp==0)
        {
            document.getElementById(eid).style.visibility="hidden"
        }
        return;
    }
    document.getElementById(eid).style.opacity=newOp;
    setTimeout("fade( '" + eid + "'," + initOp + "," + finalOp + "," + TimeToFade + "," + time + ")", TimeToFade/100);
    }
</script>
